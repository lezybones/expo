  const [screen, setScreen] = useState("home");
  const [selectedService, setSelectedService] = useState(null);
  const [selectedPlace, setSelectedPlace] = useState(null);
  const [search, setSearch] = useState("");
  const [bookingVisible, setBookingVisible] = useState(false);
  const [bookingName, setBookingName] = useState("");
  const [bookingDate, setBookingDate] = useState("");
  const [bookingPhone, setBookingPhone] = useState("");
  const [bookings, setBookings] = useState([]);

  const filteredPlaces = useMemo(() => {
    const term = search.trim().toLowerCase();

    if (!term) return PLACES;

    return PLACES.filter(
      (place) =>
        place.name.toLowerCase().includes(term) ||
        place.category.toLowerCase().includes(term)
    );
  }, [search]);

  function openService(service) {
    if (service.id === "road") {
      setScreen("road");
      return;
    }

    setSelectedService(service);
    setScreen("services");
  }

  function openPlace(place) {
    setSelectedPlace(place);
    setScreen("details");
  }

  function createBooking() {
    if (!bookingName || !bookingPhone || !bookingDate) {
      Alert.alert(
        "Complete booking",
        "Please enter your name, phone number and preferred date."
      );
      return;
    }

    const booking = {
      id: Date.now(),
      place: selectedPlace?.name || "My Paushe Service",
      name: bookingName,
      phone: bookingPhone,
      date: bookingDate,
      status: "Pending",
    };

    setBookings((current) => [booking, ...current]);

    setBookingVisible(false);
    setBookingName("");
    setBookingPhone("");
    setBookingDate("");

    Alert.alert(
      "Booking received",
      "Your My Paushe booking request has been created."
    );
  }

    Alert.alert(
      "Road assistance",
      "Your roadside assistance request has been started. The production version will connect you with verified roadside partners."
    );
  }

    Alert.alert(
      "M-Pesa",
      "The M-Pesa button is ready for integration. The production version will connect this to M-Pesa Daraja."
    );
  }
